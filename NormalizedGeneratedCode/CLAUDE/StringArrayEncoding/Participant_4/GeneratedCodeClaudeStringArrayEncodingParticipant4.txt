public class MatrixException : Exception{ public MatrixException() : base() { }}
public class MatrixString{ private string[, ] m; public MatrixString(int rows, int columns, string value) {  if (rows <= 0 || columns <= 0)  {   throw new ArgumentException();  }  m = new string[rows, columns];  for (int i = 0; i < rows; i++)  {   for (int j = 0; j < columns; j++)   {    m[i, j] = value;   }  } } public void Set(int row, int column, string value) {  if (row < 0 || row >= m.GetLength(0) || column < 0 || column >= m.GetLength(1))  {   throw new MatrixException();  }  m[row, column] = value; } public string RowToString(int index, string separator) {  if (index < 0 || index >= m.GetLength(0) || separator == null)  {   throw new MatrixException();  }  string result = string.Empty;  int columns = m.GetLength(1);  for (int j = 0; j < columns; j++)  {   result += m[index, j];   if (j < columns - 1)   {    result += separator;   }  }  return result; }}
