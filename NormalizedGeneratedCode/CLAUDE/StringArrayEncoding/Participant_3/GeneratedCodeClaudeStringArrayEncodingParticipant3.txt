public class MatrixString{ private string[, ] m; public MatrixString(int rows, int cols, string val) {  if (rows <= 0 || cols <= 0)   throw new ArgumentException("Rows and cols must be positive");  m = new string[rows, cols];  for (int i = 0; i < rows; i++)   for (int j = 0; j < cols; j++)    m[i, j] = val; } public void Set(int r, int c, string val) {  if (r < 0 || r >= m.GetLength(0) || c < 0 || c >= m.GetLength(1))   throw new MatrixException("Invalid row or column index");  m[r, c] = val; } public string RowToString(int index, string separator) {  if (separator == null)   throw new MatrixException("Separator cannot be null");  if (index < 0 || index >= m.GetLength(0))   throw new MatrixException("Invalid row index");  string result = m[index, 0];  for (int j = 1; j < m.GetLength(1); j++)   result += separator + m[index, j];  return result; }}public class MatrixException : Exception{ public MatrixException(string message) : base(message) { }}
