public class Program{ public static void Main() {  int[] A =  {   1,   2,   3,   4,   5,   10,   12  };  int[] B =  {   4,   5,   6,   7,   8,   10  };  Console.WriteLine("Array A: " + string.Join(", ", A));  Console.WriteLine("Array B: " + string.Join(", ", B));  Console.WriteLine("--------------------------------------------");  int[] resultadoLinq = EncontrarDiferencaComLinq(A, B);  Console.WriteLine("Resultado com LINQ (elementos em A que não estão em B):");  Console.WriteLine(string.Join(", ", resultadoLinq));  Console.WriteLine();  int[] resultadoLoop = EncontrarDiferencaComLoop(A, B);  Console.WriteLine("Resultado com Loop (elementos em A que não estão em B):");  Console.WriteLine(string.Join(", ", resultadoLoop)); } public static int[] EncontrarDiferencaComLinq(int[] a, int[] b) {  return a.Except(b).ToArray(); } public static int[] EncontrarDiferencaComLoop(int[] a, int[] b) {  var setB = new HashSet<int>(b);  var resultado = new List<int>();  foreach (int numeroA in a)  {   if (!setB.Contains(numeroA))   {    resultado.Add(numeroA);   }  }  return resultado.ToArray(); }}
