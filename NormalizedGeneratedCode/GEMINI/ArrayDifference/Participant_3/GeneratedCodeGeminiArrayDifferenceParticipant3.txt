public class ArrayOperations{ public static int[] Difference(int[] a, int[] b) {  return a.Except(b).ToArray(); } public static int[] Difference_Manual(int[] a, int[] b) {  HashSet<int> elementsInB = new HashSet<int>(b);  List<int> result = new List<int>();  foreach (int element in a)  {   if (!elementsInB.Contains(element))   {    result.Add(element);   }  }  return result.ToArray(); }}
