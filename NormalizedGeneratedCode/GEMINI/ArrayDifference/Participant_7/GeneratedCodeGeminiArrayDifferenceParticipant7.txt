public class ArrayDifference{ public static int[] Difference(int[] a, int[] b) {  return a.Except(b).ToArray(); } public static void Main(string[] args) {  int[] arrayA =  {   1,   2,   3,   4,   5,   6  };  int[] arrayB =  {   4,   5,   6,   7,   8  };  int[] result = Difference(arrayA, arrayB);  Console.WriteLine("Resultado: [" + string.Join(", ", result) + "]");  int[] arrayC =  {   1,   2,   2,   3,   1  };  int[] arrayD =  {   2  };  int[] result2 = Difference(arrayC, arrayD);  Console.WriteLine("Resultado 2: [" + string.Join(", ", result2) + "]"); }}
