public class ArrayUtils{ public static int[] ElementosExclusivos(int[] primeiroArray, int[] segundoArray) {  return primeiroArray.Except(segundoArray).ToArray(); } public static void Main(string[] args) {  int[] array1 =  {   1,   2,   3,   4,   5,   6  };  int[] array2 =  {   2,   4,   6,   8,   10  };  int[] resultado = ElementosExclusivos(array1, array2);  Console.WriteLine("Elementos do primeiro array que não estão no segundo:");  Console.WriteLine(string.Join(", ", resultado)); }}
