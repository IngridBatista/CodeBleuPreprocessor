public class ArrayUtils{ public static int[] ElementosUnicosEmA(int[] a, int[] b) {  return a.Except(b).ToArray(); } public static void Main(string[] args) {  int[] arrayA =  {   1,   2,   3,   4,   5,   8  };  int[] arrayB =  {   3,   5,   6,   7,   8  };  int[] resultado = ElementosUnicosEmA(arrayA, arrayB);  Console.WriteLine("Elementos de A que não estão em B: ");  Console.WriteLine(string.Join(", ", resultado)); }}
