public class MatrixException : Exception{ public MatrixException() : base("Ocorreu um erro na operação da matriz. Verifique os índices fornecidos.") { } public MatrixException(string message) : base(message) { } public MatrixException(string message, Exception innerException) : base(message, innerException) { }}
public class MatrixString{ private readonly string[, ] _matrix; public int Rows => _matrix.GetLength(0); public int Columns => _matrix.GetLength(1); public MatrixString(int rows, int columns, string value) {  if (rows <= 0 || columns <= 0)  {   throw new ArgumentException("As dimensões da matriz (linhas e colunas) devem ser maiores que zero.");  }  _matrix = new string[rows, columns];  FillMatrix(value); } public void Set(int row, int column, string value) {  ValidateIndices(row, column);  _matrix[row, column] = value; } public string RowToString(int rowIndex, string separator) {  if (separator is null)  {   throw new MatrixException("O separador não pode ser nulo.");  }  ValidateRowIndex(rowIndex);  IEnumerable<string> rowElements = GetRowElements(rowIndex);  return string.Join(separator, rowElements); } private void FillMatrix(string value) {  for (int i = 0; i < Rows; i++)  {   for (int j = 0; j < Columns; j++)   {    _matrix[i, j] = value;   }  } } private void ValidateIndices(int row, int column) {  ValidateRowIndex(row);  ValidateColumnIndex(column); } private void ValidateRowIndex(int rowIndex) {  if (rowIndex < 0 || rowIndex >= Rows)  {   throw new MatrixException($"O índice da linha ({rowIndex}) está fora dos limites válidos [0-{Rows - 1}].");  } } private void ValidateColumnIndex(int columnIndex) {  if (columnIndex < 0 || columnIndex >= Columns)  {   throw new MatrixException($"O índice da coluna ({columnIndex}) está fora dos limites válidos [0-{Columns - 1}].");  } } private IEnumerable<string> GetRowElements(int rowIndex) {  for (int j = 0; j < Columns; j++)  {   yield return _matrix[rowIndex, j];  } }}
