public class MatrixException : Exception{ public MatrixException() : base() { }}
public class MatrixString{ private readonly string[, ] m; public int Rows { get; private set; } public int Columns { get; private set; } public MatrixString(int rows, int columns, string value) {  if (rows <= 0 || columns <= 0)  {   throw new ArgumentException("As dimensões da matriz devem ser positivas.");  }  this.Rows = rows;  this.Columns = columns;  this.m = new string[rows, columns];  for (int i = 0; i < rows; i++)  {   for (int j = 0; j < columns; j++)   {    m[i, j] = value;   }  } } public void Set(int row, int column, string value) {  bool isRowInvalid = row < 0 || row >= this.Rows;  bool isColumnInvalid = column < 0 || column >= this.Columns;  if (isRowInvalid || isColumnInvalid)  {   throw new MatrixException();  }  m[row, column] = value; } public string RowToString(int index, string separator) {  if (index < 0 || index >= this.Rows)  {   throw new MatrixException();  }  if (separator == null)  {   throw new MatrixException();  }  StringBuilder sb = new StringBuilder();  for (int j = 0; j < this.Columns; j++)  {   sb.Append(m[index, j]);   if (j < this.Columns - 1)   {    sb.Append(separator);   }  }  return sb.ToString(); }}
