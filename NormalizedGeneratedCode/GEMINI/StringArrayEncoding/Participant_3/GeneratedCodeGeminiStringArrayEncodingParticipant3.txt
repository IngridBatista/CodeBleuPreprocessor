public class MatrixException : Exception{ public MatrixException(string message) : base(message) { }}public class MatrixString{ private string[, ] m; public MatrixString(int rows, int cols, string val) {  if (rows <= 0 || cols <= 0)  {   throw new ArgumentException("As dimensões da matriz devem ser positivas.");  }  this.m = new string[rows, cols];  for (int i = 0; i < rows; i++)  {   for (int j = 0; j < cols; j++)   {    this.m[i, j] = val;   }  } } public void Set(int r, int c, string val) {  if (r < 0 || r >= m.GetLength(0) || c < 0 || c >= m.GetLength(1))  {   throw new MatrixException("Índice de linha ou coluna fora dos limites da matriz.");  }  this.m[r, c] = val; } public string RowToString(int index, string separator) {  if (index < 0 || index >= m.GetLength(0))  {   throw new MatrixException("Índice de linha inválido.");  }  if (separator == null)  {   throw new MatrixException("O separador não pode ser nulo.");  }  var columnIndices = Enumerable.Range(0, m.GetLength(1));  var rowElements = columnIndices.Select(col => m[index, col]);  return string.Join(separator, rowElements); }}
