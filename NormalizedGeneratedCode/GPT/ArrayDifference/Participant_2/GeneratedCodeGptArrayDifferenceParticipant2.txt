class Program{ static void Main() {  int[] A =  {   1,   2,   3,   4,   5,   6  };  int[] B =  {   4,   5,   7,   8  };  int[] resultadoLinq = A.Where(x => !B.Contains(x)).ToArray();  Console.WriteLine("Resultado com LINQ: " + string.Join(", ", resultadoLinq));  int[] resultadoLoop = DiferencaComLoop(A, B);  Console.WriteLine("Resultado com loop: " + string.Join(", ", resultadoLoop)); } static int[] DiferencaComLoop(int[] A, int[] B) {  var conjuntoB = B.ToHashSet();  var listaResultado = new System.Collections.Generic.List<int>();  foreach (var valor in A)  {   if (!conjuntoB.Contains(valor))   {    listaResultado.Add(valor);   }  }  return listaResultado.ToArray(); }}
