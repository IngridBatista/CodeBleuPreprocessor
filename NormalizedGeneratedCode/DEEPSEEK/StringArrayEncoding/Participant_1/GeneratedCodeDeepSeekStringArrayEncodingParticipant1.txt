public class MatrixException : Exception{ public MatrixException() : base() { } public MatrixException(string message) : base(message) { }}public class MatrixString{ private string[, ] m; public int Rows { get; private set; } public int Columns { get; private set; } public MatrixString(int rows, int columns, string value) {  if (rows <= 0 || columns <= 0)  {   throw new ArgumentException();  }  this.Rows = rows;  this.Columns = columns;  m = new string[rows, columns];  for (int i = 0; i < rows; i++)  {   for (int j = 0; j < columns; j++)   {    m[i, j] = value;   }  } } public void Set(int row, int column, string value) {  if (row < 0 || row >= Rows || column < 0 || column >= Columns)  {   throw new MatrixException();  }  m[row, column] = value; } public string RowToString(int index, string separator) {  if (index < 0 || index >= Rows)  {   throw new MatrixException();  }  if (separator == null)  {   throw new MatrixException();  }  string[] rowElements = new string[Columns];  for (int j = 0; j < Columns; j++)  {   rowElements[j] = m[index, j];  }  return string.Join(separator, rowElements); }}
