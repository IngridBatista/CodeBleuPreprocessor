public class Program{ public static int[] ElementosExclusivos(int[] a, int[] b) {  HashSet<int> conjuntoB = new HashSet<int>(b);  List<int> resultado = new List<int>();  foreach (int elemento in a)  {   if (!conjuntoB.Contains(elemento))   {    resultado.Add(elemento);   }  }  return resultado.ToArray(); } public static void Main() {  int[] a =  {   1,   2,   3,   4,   5,   6  };  int[] b =  {   2,   4,   6  };  int[] resultado = ElementosExclusivos(a, b);  Console.WriteLine("Elementos em a que não estão em b:");  foreach (int num in resultado)  {   Console.Write(num + " ");  } }}
