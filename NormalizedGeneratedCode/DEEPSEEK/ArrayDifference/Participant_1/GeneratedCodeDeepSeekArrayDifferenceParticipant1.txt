public class Program{ public static int[] DiferencaEntreArrays(int[] array1, int[] array2) {  return array1.Where(x => !array2.Contains(x)).ToArray(); } public static int[] DiferencaEntreArraysOtimizado(int[] array1, int[] array2) {  var set2 = new HashSet<int>(array2);  return array1.Where(x => !set2.Contains(x)).ToArray(); } public static int[] DiferencaEntreArraysSemLINQ(int[] array1, int[] array2) {  var resultado = new List<int>();  foreach (var item in array1)  {   bool encontrado = false;   foreach (var item2 in array2)   {    if (item == item2)    {     encontrado = true;     break;    }   }   if (!encontrado)   {    resultado.Add(item);   }  }  return resultado.ToArray(); } public static void Main() {  int[] array1 =  {   1,   2,   3,   4,   5,   6  };  int[] array2 =  {   2,   4,   6  };  int[] diferenca = DiferencaEntreArrays(array1, array2);  Console.WriteLine("Resultado: " + string.Join(", ", diferenca));  int[] diferencaOtimizada = DiferencaEntreArraysOtimizado(array1, array2);  Console.WriteLine("Resultado otimizado: " + string.Join(", ", diferencaOtimizada)); }}
